```sql
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,                    -- 'cup_coffee_white', 'suitcase_brown', 'bed_queen'

  width INTEGER NOT NULL,
  height INTEGER NOT NULL,
  spanish_terms TEXT[],                  -- ['taza', 'taza de café']
  english_terms TEXT[],                  -- ['cup', 'coffee cup']
  tags TEXT[],                           -- ['kitchen', 'beverage', 'ceramic']
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Pre-composed scene templates (common arrangements)
CREATE TABLE visual_compositions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,                    -- 'hotel_checkin_scene', 'cafe_ordering'
  display_name TEXT NOT NULL,
  environment_id UUID REFERENCES visual_environments(id),

  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Link vocabulary to visual assets (many-to-many)
CREATE TABLE vocabulary_visual_assets (
  vocabulary_id UUID REFERENCES vocabulary(id),
```

### Handler Logic (Pseudo-code)

```typescript
// In gemini-function-handlers.ts

async function handle_compose_visual_scene(params: ComposeVisualParams, context: SessionContext) {
  const { environment, objects, preposition_context, fallback_to_generation = true } = params;
  
  // 1. Check if we have a pre-composed template that matches this exactly
  const template = await findMatchingComposition({
    environment,
    objects: objects.map(o => o.term),
    preposition_context
  });
  
  if (template) {
    await logUsage(template.id);
    return {
      success: true,
      image_url: template.image_url,
      source: 'pre_composed_template',
      composition_id: template.id,
      cache_hit: true
    };
  }
  
  // 2. Try to compose on-demand from our asset library
  const baseEnvironment = await getEnvironment(environment);
  if (!baseEnvironment) {
    if (!fallback_to_generation) {
      return { success: false, error: `Environment '${environment}' not found in library` };
    }
    // Fall through to generation
  } else {
    const assetMatches = await Promise.all(
      objects.map(obj => findBestAsset(obj.term, { environment, context: preposition_context }))
    );
    
    if (assetMatches.every(a => a !== null)) {
      // We have all the assets we need — compose the scene
      const composedImage = await compositeScene({
        base: baseEnvironment,
        objects: assetMatches.map((asset, i) => ({
          asset,
          position: objects[i].position || 'center',
          emphasis: objects[i].emphasis || false
        }))
      });
      
      // Upload composed result to GCS
      const imageUrl = await uploadToGCS(composedImage, 'composed_visuals');
      
      // Optionally cache this composition for reuse
      await cacheComposition({
        environment,
        objects,
        preposition_context,
        image_url: imageUrl
      });
      
      return {
        success: true,
        image_url: imageUrl,
        source: 'dynamic_composition',
        cache_hit: false
      };
    }
  }
  
  // 3. Fallback: Use DALL-E generation (your existing system)
  if (fallback_to_generation) {
    const prompt = buildPromptFromComposition(environment, objects, preposition_context);
    return await handle_generate_visual_aid({ prompt, style: 'educational' }, context);
  }
  
  return { 
    success: false, 
    error: 'Could not compose scene and fallback disabled' 
  };
}
```

### Compositing Service

```typescript
// New file: server/services/image-compositor.ts

import sharp from 'sharp';  // Popular Node.js image processing library

interface CompositeLayer {
  asset: VisualAsset;
  position: string;  // 'center', 'left', 'on_table', etc.
  emphasis: boolean;
}

async function compositeScene(config: {
  base: VisualEnvironment;
  objects: CompositeLayer[];
}): Promise<Buffer> {
  
  const { base, objects } = config;
  
  // Download base environment
  const baseImage = await downloadImage(base.image_url);
  
  // Start with base layer
  let composite = sharp(baseImage);
  const layers = [];
  
  for (const obj of objects) {
    // Download object asset
    const objectImage = await downloadImage(obj.asset.image_url);
    
    // Calculate position based on position string and base dimensions
    const { x, y, scale } = calculatePosition(
      obj.position,
      base.width,
      base.height,
      obj.asset.width,
      obj.asset.height
    );
    
    // Resize if needed
    let processedObject = sharp(objectImage);
    if (scale !== 1.0) {
      processedObject = processedObject.resize({
        width: Math.round(obj.asset.width * scale),
        height: Math.round(obj.asset.height * scale)
      });
    }
    
    // Add emphasis effect (glow, outline, etc.)
    if (obj.emphasis) {
      processedObject = await addEmphasis(processedObject);
    }
    
    layers.push({
      input: await processedObject.toBuffer(),
      top: y,
      left: x
    });
  }
  
  // Composite all layers
  const result = await composite
    .composite(layers)
    .png()
    .toBuffer();
  
  return result;
}

function calculatePosition(
  positionHint: string,
  baseWidth: number,
  baseHeight: number,
  objectWidth: number,
  objectHeight: number
): { x: number; y: number; scale: number } {
  
  // Simple position mapping (you'd refine this based on your environments)
  const positions = {
    'center': { x: 0.5, y: 0.5, scale: 1.0 },
    'left': { x: 0.25, y: 0.5, scale: 0.8 },
    'right': { x: 0.75, y: 0.5, scale: 0.8 },
    'foreground': { x: 0.5, y: 0.7, scale: 1.2 },
    'background': { x: 0.5, y: 0.3, scale: 0.6 },
    'on_table': { x: 0.5, y: 0.65, scale: 0.7 },
    'on_floor': { x: 0.5, y: 0.85, scale: 0.5 },
    // ... more position presets
  };
  
  const pos = positions[positionHint] || positions['center'];
  
  return {
    x: Math.round((baseWidth * pos.x) - (objectWidth * pos.scale / 2)),
    y: Math.round((baseHeight * pos.y) - (objectHeight * pos.scale / 2)),
    scale: pos.scale
  };
}

async function addEmphasis(image: sharp.Sharp): Promise<sharp.Sharp> {

  // Could use sharp's composite with a glow layer, or color manipulation
  // Simplified example:
  return image.modulate({ brightness: 1.1 });
}
```
