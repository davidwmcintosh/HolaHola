        TargetPath = $null
        TargetKind = $null
        RequestKey = $null
        SavedState = $null
    }

        if (-not [IO.File]::Exists($RuntimeRequestPath)) {
            throw 'diagnostic precondition failed'

        $saved = Read-DpapiJson -Path $RuntimeRequestPath `
            -FailureCode 'runtime_request_corrupted'

            'endpoint', 'requestKey', 'issueId', 'manifest', 'installed',
            'ackPayload', 'ackSignature'

        $diag.SavedState = $saved

        function Get-SafeBootstrapPostFailure {
            param([Parameter(Mandatory = $true)]$ErrorRecord)

            $current = $null
            $exceptionProperty = $ErrorRecord.PSObject.Properties['Exception']
            if ($null -ne $exceptionProperty) {
                $current = $exceptionProperty.Value
            }

            while ($null -ne $current) {
                $responseProperty = $current.PSObject.Properties['Response']
                if ($null -ne $responseProperty -and
                    $null -ne $responseProperty.Value) {
                    $response = $responseProperty.Value
                    $statusProperty = $response.PSObject.Properties['StatusCode']

                    if ($null -ne $statusProperty -and
                        $null -ne $statusProperty.Value) {
                        $status = [int]$statusProperty.Value
                        $body = $null

                        $detailsProperty = $ErrorRecord.PSObject.Properties['ErrorDetails']
                        if ($null -ne $detailsProperty -and
                            $null -ne $detailsProperty.Value) {
                            $messageProperty = $detailsProperty.Value.PSObject.Properties['Message']
                            if ($null -ne $messageProperty) {
                                $body = [string]$messageProperty.Value
                            }
                        }

                        if ([string]::IsNullOrWhiteSpace($body)) {
                            $streamMethod = $response.PSObject.Methods['GetResponseStream']
                            if ($null -ne $streamMethod) {
                                try {
                                    $stream = $response.GetResponseStream()
                                    if ($null -ne $stream) {
                                        $reader = New-Object System.IO.StreamReader($stream)
                                        try { $body = $reader.ReadToEnd() }
                                        finally { $reader.Dispose() }
                                    }
                                } catch {
                                    $body = $null
                                }
                            }
                        }

                        $code = 'unavailable'
                        if (-not [string]::IsNullOrWhiteSpace($body)) {
                            try {
                                $payload = $body | ConvertFrom-Json -ErrorAction Stop
                                $candidate = $null
                                $codeProperty = $payload.PSObject.Properties['code']

                                if ($null -ne $codeProperty) {
                                    $candidate = [string]$codeProperty.Value
                                } else {
                                    $errorProperty = $payload.PSObject.Properties['error']
                                    if ($null -ne $errorProperty) {
                                        if ($errorProperty.Value -is [string]) {
                                            $candidate = [string]$errorProperty.Value
                                        } elseif ($null -ne $errorProperty.Value) {
                                            $nestedCode = $errorProperty.Value.PSObject.Properties['code']
                                            if ($null -ne $nestedCode) {
                                                $candidate = [string]$nestedCode.Value
                                            }
                                        }
                                    }
                                }

                                if ($candidate -cmatch '^[A-Za-z0-9][A-Za-z0-9_.-]{0,79}$') {
                                    $code = $candidate
                                }
                            } catch {
                                $code = 'unavailable'
                            }
                        }

                        return ('HTTP status=' + $status + '; errorCode=' + $code)
                    }
                }

                if ($current -is [System.Net.WebException]) {
                    return ('networkExceptionType=' + $current.GetType().Name +
                        '; networkStatus=' + [string]$current.Status)
                }

                $innerProperty = $current.PSObject.Properties['InnerException']
                if ($null -eq $innerProperty) { break }
                $current = $innerProperty.Value
            }

            if ($null -eq $current) {
                return 'networkExceptionType=unknown; networkStatus=unknown'
            }
            return ('networkExceptionType=' + $current.GetType().Name +
                '; networkStatus=unknown')
        }

        function Invoke-HostAuthenticatedRequest {
            param(
                [Parameter(Mandatory = $true)][string]$Method,
                [Parameter(Mandatory = $true)][string]$Endpoint,
                [Parameter(Mandatory = $true)][string]$Path,
                [Parameter(Mandatory = $true)]$Identity,
                [AllowEmptyString()][string]$Body = ''
            )

            # Block every request except the single route selected from saved state.
            if ($Method -cne 'POST' -or
                $Endpoint.TrimEnd('/') -cne 'https://getholahola.com' -or
                $Path -cne $diag.TargetPath) {
                throw 'diagnostic scope guard'
            }

            # Verify the initializer supplied the already-persisted key/payload.
            $wire = $Body | ConvertFrom-Json -ErrorAction Stop
            if ($diag.TargetKind -eq 'issue') {
                Assert-ExactPropertySet -Value $wire -Names @(
                    'requestKey', 'protocolVersion'
                ) -FailureCode 'diagnostic_payload_mismatch'
                if ([string]$wire.requestKey -cne $diag.RequestKey -or
                    [int]$wire.protocolVersion -ne 1) {
                    throw 'diagnostic payload guard'
                }
            } else {
                Assert-ExactPropertySet -Value $wire -Names @(
                    'payload', 'signature'
                ) -FailureCode 'diagnostic_payload_mismatch'
                if ([string]$wire.signature -cne [string]$diag.SavedState.ackSignature -or
                    (ConvertTo-CanonicalJson -Value $wire.payload) -cne
                    (ConvertTo-CanonicalJson -Value $diag.SavedState.ackPayload)) {
                    throw 'diagnostic payload guard'
                }
            }

            $signature = $Identity.rsa.SignData(
                [Text.Encoding]::UTF8.GetBytes([string]$Identity.accessToken),
                [Security.Cryptography.CryptoConfig]::MapNameToOID('SHA256'))

            $headers = @{
                'x-coordination-v2-host-token' = [string]$Identity.accessToken
                'x-coordination-v2-host-proof' = [Convert]::ToBase64String($signature)
            }

            try {
                # Same request method and options as the launcher.
                $null = Invoke-RestMethod -Method $Method `
                    -Uri ($Endpoint.TrimEnd('/') + $Path) `
                    -Headers $headers -ContentType 'application/json' -Body $Body `
                    -UseBasicParsing -MaximumRedirection 0 -ErrorAction Stop

                $diag.Summary = 'method=POST route=' + $Path +
                    ' HTTP status=success (exact status unavailable from Invoke-RestMethod); errorCode=none'
                throw 'diagnostic stop after target POST'
            } catch {
                # Capture the original Invoke-RestMethod error directly here.
                $originalError = $_
                $diag.Summary = 'method=POST route=' + $Path + ' ' +
                    (Get-SafeBootstrapPostFailure -ErrorRecord $originalError)
                throw 'diagnostic stop after target POST'
            }
        }

        # The initializer retains its existing security checks, identity,
        # request key, payload construction, and signing flow.
        $null = Initialize-HolaCoordinatorRuntime -Endpoint 'https://getholahola.com'
    } catch {
        if (-not [string]::IsNullOrWhiteSpace([string]$diag.Summary)) {
            $diag.Summary
        } else {
            'Diagnostic stopped before the target POST; no request result.'
        }
    }
}
